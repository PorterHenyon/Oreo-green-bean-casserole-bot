## Oreo bot (PowerShell)
## Copy/paste the commands below into a terminal (any folder).

# 0) Ensure we're in this repo folder
Set-Location -Path "$env:USERPROFILE\.vscode\Oreo-green-bean-casserole-bot"

# 1) (Optional) create a virtual environment once
python -m venv .venv

# 2) Activate it (each new terminal session)
.\.venv\Scripts\Activate.ps1

# 3) Install/update dependencies
python -m pip install -r requirements.txt

# 4) Run the bot
python main.py
